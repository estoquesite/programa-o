function analisarScript() { const textoScript = document.getElementById('inputScriptTexto').value; const textoNotas = document.getElementById('inputNotasSala').value; const textoCombinado = textoScript + "\n" + textoNotas; const containerSugestoes = document.getElementById('resultadoSugestoes'); containerSugestoes.innerHTML = ''; if (!textoCombinado.trim()) { containerSugestoes.innerHTML = `

Insira algum texto na transcrição ou nas notas da sala.

`; return; } const palavrasChave = JSON.parse(localStorage.getItem('programacao_palavras_chave_custom')) || palavrasChavePadrao; // Divide o texto em parágrafos ou blocos de fala completos para trazer o contexto inteiro const blocos = textoCombinado.split(/\n+/); let encontradas = 0; blocos.forEach(bloco => { const blocoTrim = bloco.trim(); const blocoUp = blocoTrim.toUpperCase(); if (blocoTrim.length > 0) { // Verifica se alguma das palavras-chave está presente no bloco de texto const contemPalavra = palavrasChave.some(p => { // Cria uma busca flexível por palavra inteira ou trecho const regex = new RegExp(p, 'i'); return regex.test(blocoUp); }); if (contemPalavra) { encontradas++; const div = document.createElement('div'); div.className = 'flex flex-col gap-1.5 bg-blue-50 dark:bg-blue-950/40 p-2.5 rounded-lg border border-blue-200 dark:border-blue-900'; div.innerHTML = ` 💬 "${blocoTrim}"
`; containerSugestoes.appendChild(div); } } }); if (encontradas === 0) { containerSugestoes.innerHTML = `

Nenhuma palavra-chave reconhecida no texto combinado. Verifique se cadastrou o termo em "Gerir Palavras".

`; } }
